267:  Приложение 2. Скрипт расчета KPI и генерации аналитических отчетов (Python)
269: import pandas as pd
270: import matplotlib.pyplot as plt
271: from sqlalchemy.orm import Session
272: from database import engine
273: from models import Task
275: def generate_kpi_report(db: Session):
276:     # Извлечение данных из спроектированной БД
277:     tasks = pd.read_sql(db.query(Task).statement, db.bind)
279:     if tasks.empty:
280:         return "Нет данных для анализа"
282:     # Расчет метрик
283:     total = len(tasks)
284:     completed = len(tasks[tasks['status'] == 'done'])
285:     completion_rate = (completed / total) * 100
287:     # Визуализация KPI (Matplotlib)
288:     plt.figure(figsize=(8, 5))
289:     status_counts = tasks['status'].value_counts()
290:     colors = ['#8A2BE2' if s == 'done' else '#FF9500' for s in status_counts.index]
292:     status_counts.plot(kind='bar', color=colors, edgecolor='black')
293:     plt.title('Распределение задач по статусам (KPI отдела)')
294:     plt.xlabel('Статус задачи')
295:     plt.ylabel('Количество')
296:     plt.xticks(rotation=0)
297:     plt.grid(axis='y', linestyle='--', alpha=0.7)
299:     plt.tight_layout()
300:     plt.savefig('kpi_report.png')
301:     plt.close()
303:     return f"Процент выполнения: {completion_rate:.1f}%"
